# 一维数组

- 定义

  > 数据类型 数组名[元素个数];

- 数组的初始化

  ```c
  //声明时显式初始化
  int arr[5] = {1, 2, 3, 4, 5};
  
  //部分初始化
  int arr[5] = {1, 2};
  
  //不指定数组大小，由编译器推断
  int arr[] = {1, 2, 3, 4, 5};
  
  //使用for循环逐一赋值
  int arr[5];
  for (int i= 0; i< 5; i++) {
  	arr[i] = i+ 1;
  }
  
  //静态初始化
  //在声明静态数组时，未显式初始化的元素会自动初始化为0。
  static int arr[5];
  
  //全部初始化为0
  int arr[5] = {0};
  ```

- 访问数组元素

  ```c
  //通过下标（索引）来访问数组的每个元素，数组下标从0开始。
  arr[i] = 10;// 将10赋值给数组的第一个元素
  int *p = arr;	//这里p和arr等价
  *(p + i);		//等价于arr[i]
  *(arr + i);		//等价于arr[i]
  p[i];			//等价于arr[i]
  *(p++);			//当前等价与arr[i], 之后为arr[i+1]
  i[arr];			//arr[i] = *(arr+i) = *(i+arr) = i[arr]
  i[p];
  
  //arr是指针常量, 其指向的地址不可变
  //p是指针变量, 其指向的地址可以变化
  ```

- 数组遍历

  ```c
  for(inti= 0; i< 5; i++) {
  	printf("%d ", numbers[i]);
  }
  ```

- 数组的大小和长度

  ```c
  sizeof(numbers)/sizeof(int)
  ```

- 数组指针

  ```c
  //数组元素在内存中是连续存储的，因此可以通过指针访问相邻元素。假设数组int numbers[5]存储在内存中，其首元素numbers[0]的地址为&numbers[0]，那么numbers[1]的地址为&numbers[0] + 1。数组名可以直接当作指针使用，通过指针访问数组元素：
  intnumbers[5] = {10, 20, 30, 40, 50};
  int*ptr= numbers;
  printf("%d", *(ptr+ 1));// 输出numbers[1]的值
  ```

- 数组越界

  ```c
  //访问超过数组大小范围的索引（数组越界）是危险的，可能导致程序崩溃或意外行为。
  printf("%d", numbers[10]); // 越界访问，行为未定义
  ```

# 字符数组

- 定义

  > 字符数组是用于存储字符序列的数组，每个元素是一个字符类型（char）。
  > 字符数组通常用于表示字符串，但本质上它只是字符的集合。

- 声明语法

  ```c
  char array_name[size];
  示例：
  char name[10]; // 声明一个长度为10的字符数组
  ```

- 初始化

  ```c
  //显式初始化
  char name[5] = {'H', 'e', 'l', 'l', 'o'};
  
  //字符串初始化
  char greeting[] = "Hello"; //自动在末尾添加字符串结束符'\0'
  char str[5] = {'h', 'e', 'l', 'l', 'o'};	//初始化有问题,缺少字符串终止符
  ```

- 字符数组的赋值

  ```c
  //不可直接赋值：字符串数组不能像基本数据类型那样通过直接赋值的方式进行赋值。例如，下面的代码是错误的：
  char str[20];
  str = "Hello";// 错误，不能直接给字符数组赋值
  ```

  1. 使用strcpy()函数

     ```c
     char str[20];
     char *p = "Hello World";	//结尾自动添加'\0'
     strcpy(str, p);
     ```

     **'\0'**在C语言中的ASCII码值为0

     ![strcpy赋值](/assets/pictures/strcpy赋值.png)
     > 初始化str为全z, strcpy函数拷贝字符串后, 自动在结尾添加\0
     
  2. 使用sprintf()函数
  
     ```c
     char str[20];
     int i = 10;
     sprintf(str, "%d", i);		//sprintf接受三个参数, str为目标地址, 后边和printf类似为格式化参数, 最终会将结果以字符串形式保存, 添加'\0'
     ```
     
     ![sprintf赋值](/assets/pictures/sprintf赋值.png)
  3. 使用memcpy()函数
  
     ```c
     char str[];
     char i = 'a';
     memcpy(str, "hello", 6);	//需要考虑字符串结束'\0'
     memcpy(str, &i, 1);			//memcpy接受三个参数, str为目标地址, 后边跟原始地址和长度
     ```
  
     ![memcpy赋值](/assets/pictures/memcpy赋值.png)

- 字符数组常见问题
  1. 字符数组必须有足够的空间存储所有字符以及字符串结束符\0
  2. 字符数组操作时要注意避免越界访问
  3. 处理输入中带有空格的字符串：使用gets()（已废弃）或fgets()函数来读取包含空格的字符串。

- memset函数

  ```c
  memset(arr, 0, sizeof(arr)); 	//arr: 目标起始地址, 0: 目标值, sizeof(arr): 目标地址大小
  ```

  

# 二维数组

- 二维数组是相同类型的多行多列数据的集合，类似于一个表格（矩阵）

  ```c
  type array_name[rows][cols];
  ```

- 二维数组在内存中时线性的, 存储顺序是按行优先

  > ar\[0][0] ar\[0][1] ar\[0][2] ar\[1][0] ar\[1][1] ar\[1][2] ar\[2][0] r\[2][1] ar\[2][2]

- 声明和初始化

  ```c
  int ar[3][4]; ——声明一个3行4列的整数数组，未初始化，内容为随机值。
  int ar[2][3] = { {1, 2, 3}, {4, 5, 6} };	//显式初始化
  int ar[2][3] = { {1}, {4} };				//部分初始化,其余元素初始化为0
  int ar[][3] = { {1, 2, 3}, {4, 5, 6} };	//隐式推断大小, 必须指定列数, 函数可以自行判断
  ```

- 拷贝

  ```c
  //可以使用 memcpy函数来拷贝整个二维数组的数据块。
  //示例：
  memcpy(arr2, arr1, sizeof(arr1));	//将arr1的内容拷贝到 arr2
  ```

- 二维数组与指针的关系

  > 二维数组名作为指向首行的指针
  >
  > arr是指向 arr[0] 的指针，arr[0] 又是指向第一行的指针。

```c
int *ptr= &arr[0][0];
printf("%d\n", *(ptr+ 1)); // 访问 arr[0][1]
```

# 作业

- 字符串处理函数

  ```c
  strlen();
  strcat();
  strcpy();
  strcmp();
  
  #include <stddef.h>		
  //size_t定义: 操作系统相关
  //32位系统
  //typedef unsigned int size_t;
  //64位系统
  //typedef unsigned long size_t;
  
  
  
  size_t myStrlen(const char *str);
  char *myStrcpy(char *dest, const char *src);
  char *myStrcat(char *dest, const char *src);
  int myStrcmp(const char *str1, const char *str2);
  
  
  //初始长度0, 指针和'\0'对比, 不匹配指针和长度均自加, 结束后返回长度
  size_t myStrlen(const char *str){
      size_t len = 0;
      while(*str != '\0'){
          len++;
          str++;
      }
      return len;
  }
  
  //返回目标指针, 两个指针赋值后自加, 遇到\0后表达式值为0结束
  char *myStrcpy(char *dest, const char *src){
      char *start = dest;
      while(*dest++ = *src++);
      return start;
  }
  
  //返回目标指针, 先目标指针遍历到结尾, 然后指针赋值并自加, 遇到\0后表达式值为0结束
  char *myStrcat(char *dest, const char *src){
      char *start = dest;
      while(*dest)
          dest++;
      while(*dest++ = *src++);
      return start;
  }
  
  //返回指针最后指向的数据的差值, 任意指针指向的数据都为\0或所指向的数据不相等结束
  int myStrcmp(const char *str1, const char *str2){
      while(*str1 != '\0' && *str2 != '\0' && *str1 == *str2){
          str1++;
          str2++;
      }
      return *str1 - *str2;
  }
  ```

- sizeof和strlen的区别

  | 项目       | sizeof                        | strlen                     |
  | ---------- | ----------------------------- | -------------------------- |
  | 性质       | 运算符                        | 库函数                     |
  | 计算时机   | 编译期（变长数组 VLA 运行期） | 运行时循环遍历             |
  | 计算内容   | 占用内存字节总数              | `\0`之前字符个数，不含`\0` |
  | 依赖`\0`？ | 不需要                        | 必须依靠`\0`终止           |
  | 参数       | 任意数据类型                  | char* 字符串首地址         |
  | 数组名     | 不退化（sizeof (数组名)）     | 数组名传入直接退化指针     |

- 查找数组中的特定元素

  > 输入10个整数和一个目标数, 查找该目标数是否在数组中出现, 若出现则输出其下表, 若未出现在输出未找到

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <string.h>


int main(int argc, char *argv[])
{
    int num[10] = {0};
    int target = 0;
    int count = 0;
    printf("请输入10个整数:");
    for(int i = 0; i < 10; i++)
    {
        scanf("%d", &num[i]);
    }
    printf("请输入目标数:");
    scanf("%d", &target);
    for(int i = 0; i < 10; i++)
    {
        if(num[i] == target)
        {
            count++;
            printf("%d ", i);
        }
    }
    printf("目标数在数组中出现了%d次\n", count);
    return 0;
}
```

- 数组排序

  > 输入10个整数, 对元素进行升序排序后输出

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <string.h>

int main(int argc, char *argv[])
{
    int num[10] = {0};
    int flag = 1;
    int temp = 0;
    printf("请输入10个整数:");
    for(int i = 0; i < 10; i++)
    {
        scanf("%d", &num[i]);
    }
    for(int i = 0; i < 9; i++)
    {
        for(int j = 0; j < 9 - i -1; j++)
        {
            if (num[j] > num[j+1])
            {
                temp = num[j];
                num[j] = num[j+1];
                num[j+1] = temp;
                flag = 1;
            }
        }
        if (flag == 0)
        {
            break;
        }
    }
    for(int i = 0; i < 10; i++)
    {
        printf("%d ", num[i]);
    }
    return 0;
}
```

- 数组去重

  > 10个整数数组, 将数组中的重复数去除, 并输出

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <string.h>

int main(int argc, char *argv[]){
    int arr[10] = {1, 5, 8, 8, 1, 4, 7, 5, 6, 0};
    int temp[10] = {};
    int flag = 0;
    bool isMatch = false;
    for(int i = 0; i < 10; i++){
        for(int j = 0; j < flag; j++){
            if(temp[j] == arr[i]){
                isMatch = true;
                break;
            }
        }
        if(!isMatch){
            temp[flag] = arr[i];
            flag++;
        }
        isMatch = false;
    }
    for(int i = 0; i <= flag; i++){
        printf("%d ", temp[i]);
    }
    system("pause");
}
```

# 错误分析

- 以下那些是C语言中处理数组指针的正确方法

  ```c
  int* p = arr;
  int* p = &arr[0];
  p = arr;			//通过数组名直接获取地址
  p = arr + 1;		//通过偏移量来访问数组的后续元素
  ```

- C语言访问数组元素时, 数组下标可以是0 和正整数, 负数会导致数组访问越界
