# 无名管道

无名管道是内核空间实现的机制，只能用于亲缘进程间通信，无名管道的大小是64KB

![无名管道](/../assets/pictures/无名管道.png)

## pip函数

- 头文件

  #include <unistd.h>

- 原型

  int pipe(int pipefd[2]);

- 功能

  创建单向数据通道(半双工通信方式)
  数据只能从读端pipefd[0]读取，从写端pipefd[1]写入，方向不可逆转；
  数据并非在进程间进行传递，而是先写入内核缓冲区，再由读进程从内核缓冲区读取；

- 参数

  pipefd: 用于存放管道读写两端的文件描述符的数组首地址

- 返回值

  成功：0
  失败：-1，重置错误码

## 示例代码

```c
#include <stdio.h>
#include <unistd.h>
#include <sys/types.h>
#include <string.h>
#include <stdlib.h>
#include <sys/wait.h>

int main(int argc, const char *argv[])
{
    int pipfd[2] = {0};
    int pipe_return = 0;
    pipe_return = pipe(pipfd);
    if (pipe_return == -1)
    {
        perror("Create pipe failed");
        return -1;
    }
    pid_t pid = fork();
    if (pid == -1)
    {
        perror("Create child process failed");
        return -1;
    }
    else if ( pid == 0)
    {
        //子进程
        //关闭读端
        int close_return = close(pipfd[0]);
        if( close_return == -1)
        {
            perror("Child process close read side failed");
            return -1;
        }
        char buffer[] = "hello world";
        ssize_t write_return = 0;
        write_return = write(pipfd[1], buffer, strlen(buffer));
        if (write_return == -1 )
        {
            perror("Write to pipe failed");
            return -1;
        }
        exit(EXIT_SUCCESS);
    }
    else if (pid > 0)
    {
        //父进程
        //关闭写端
        int close_return = close(pipfd[1]);
        if( close_return == -1)
        {
            perror("Parent process close write side failed");
            return -1;
        }
        //等待子进程写入完成
        wait(NULL);
        char buffer[BUFSIZ] = {0};
        int read_return = read(pipfd[0], buffer, BUFSIZ);
        if (read_return == -1)
        {
            perror("Parent process read failed");
            return -1;
        }
        printf("Message from child process:\n");
        printf("%s\n", buffer);
    }
    return 0;
}
```

## 特点

• 无名管道依托具有亲缘关系的进程间的文件描述符继承特性来实现通信；
• 数据在无名管道中只能单向传输，符合半双工的定义；
• 在多数 Linux 系统中，常见无名管道的内核缓冲区默认大小为 64KB；
• 无名管道不能使用lseek函数，lseek用于随机访问文件位置，因此调用会出错返回-1；

## 读写特点

• 若读端存在时，写管道满则阻塞；有 4KB 空间时唤醒，是linux内核为平衡效率与响应性的设计；
• 若读端不存在时，写管道会破裂，会触发SIGPIPE信号；
• 若写端存在时，读管道读取数据，没有数据的时候阻塞等待；
• 若写端不存在时，读管道读取数据，没有数据的时候立即返回0(非阻塞)；

# 有名管道

有名管道可以用于亲缘进程通信，也可以用于非亲缘进程通信，有名管道的大小是64KB；

![有名管道](/../assets/pictures/有名管道.png)

## 亲缘进程通信

## mkfifo函数

- 头文件

  #include <sys/types.h>
  #include <sys/stat.h>

- 原型

  int mkfifo(const char *pathname, mode_t mode);

- 功能

  创建有名管道

- 参数

  pathname 文件的路径和名字(只写名字则默认路径是当前路径), 如果管道文件存在，则会进行报错(提示文件已存在)
  mode 文件的权限 最终权限=mode & ~umask

- 返回值

  成功：0
  失败：-1，重置错误码

- 使用有名管道时, 不论读写, 打开管道文件都会阻塞等待另一个进程以另外一种方式打开, 否则会一直阻塞

## unlink函数

- 头文件

  #include <unistd.h>

- 原型

  int unlink(const char *pathname);

- 功能

  删除文件系统中的”文件名链接”

- 参数

  pathname 文件的路径和名字(只写名字则默认路径是当前路径)

- 返回值

  成功：0
  失败：-1，重置错误码

- 使用有名管道时, 建议使用unlink函数删除原有的管道文件链接, 防止打开管道文件失败

- 一般使用unlink函数删除管道文件在读端进行操作

  

## 示例代码:

```c
#include <stdio.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <unistd.h>
#include <fcntl.h>
#include <stdlib.h>

int main(int argc, char *argv[])
{
    if (argc != 2)
    {
        printf("Please check arguments\n");
        return -1;
    }
    if (mkfifo(argv[1], 0666) == -1)
    {
        perror("Create fifo failed");
        return -1;
    }
    pid_t pid = fork();
    if (pid == -1)
    {
        perror("Create child process faild");
        return -1;
    }
    else if (pid == 0)
    {
        // 子进程
        unlink(argv[1]);
        int write_fd = open(argv[1], O_WRONLY);
        if (write_fd == -1)
        {
            perror("Child process open fifo failed");
            return -1;
        }
        char buffer[] = "hello world\n";
        write(write_fd, buffer, sizeof(buffer));
        close(write_fd);
        exit(EXIT_SUCCESS);
    }
    else if (pid > 0)
    {
        // 父进程
        // 打开管道文件会阻塞等待其他进程以另一种方式打开
        unlink(argv[1]);
        int read_fd = open(argv[1], O_RDONLY);
        if (read_fd == -1)
        {
            perror("Parent process open fifo failed");
            return -1;
        }

        wait(NULL);
        

        char buffer[BUFSIZ] = {0};
        ssize_t read_return = 0;
        read_return = read(read_fd, buffer, sizeof(buffer));
        if (read_return == -1)
        {
            perror("Parent process read file failed");
            return -1;
        }
        printf("Parent process read success:\n");
        printf("%s", buffer);
        close(read_fd);
        unlink(argv[1]);
        exit(EXIT_SUCCESS);
    }
    return 0;
}
```

```c

#include <stdio.h>

    // 读进程

    {

    if (read_return == -1)

    }
    printf("Read process read success:\n");
    printf("%s", buffer);
    close(read_fd);
    unlink(argv[1]);
    exit(EXIT_SUCCESS);
    return 0;

// 写进程
#include <stdio.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <unistd.h>
#include <fcntl.h>
#include <stdlib.h>

// 写进程
int main(int argc, char *argv[])
{
    if (argc != 2)
    {
        printf("Please check arguments\n");
        return -1;
    }
    if (mkfifo(argv[1], 0666) == -1)

        unlink(argv[1]);
        return -1;
    }
    // 打开管道文件会阻塞等待其他进程以另一种方式打开
    int write_fd = open(argv[1], O_WRONLY);

    {
        perror("Write process open fifo failed");
        return -1;
    }

    char buffer[] = "hello world\n";
    write(write_fd, buffer, sizeof(buffer));
    close(write_fd);
    exit(EXIT_SUCCESS);
    return 0;

```

## 有名管道特点

- 有名管道特点:
  • 数据在有名管道中只能单向传输，若要双向通信，需创建两个管道，符合半双工的定义；
  • 在多数 Linux 系统中，常见有名管道的内核缓冲区默认大小为64KB；
  • 有名管道不能使用 lseek 函数，lseek用于随机访问文件位置，因此调用会出错返回 -1；

- 有名管道写操作特点：
  • 当读端已打开时：
  • 写入数据时会尽可能写入(受管道缓冲区大小限制)；
  • 若管道已满，写操作会阻塞直到有空间；
  • 若写操作在阻塞期间读端关闭，会产生SIGPIPE信号(管道破裂)；
  • 当读端未打开时：
  • 写进程会在open( )调用时会阻塞，直到有读进程打开该FIFO；
  • 若写进程以非阻塞方式打开FIFO且无读端，open( )会返回错误;

- 有名管道读操作特点

  • 当写端已打开时：
  • 读取数据时，有多少读取多少；
  • 若管道为空，读操作会阻塞等待数据写入；
  • 当写端未打开时：
  • 读进程在open( )调用会阻塞，直到有写进程打开该管道；
  • 若写端曾经打开但已关闭，读操作会读取剩下数据，读完后返回0(不会阻塞)；
  • 若读进程以非阻塞方式打开FIFO且无写端，open会返回成功；

# 信号

信号是中断的一种软件模拟，它是基于linux内核实现的

用户可以给进程发信号，进程可以给进程发信号，linux内核也可以给进程发信号

对信号处理的三种方式：默认、忽略、捕捉

## 常用信号

- SIGHUP 用户终端关闭时产生，发给与该终端关联会话内的所有进程  终止
- SIGINT 用户键入 INTR 字符(Ctrl+C)时产生，内核发送到当前终端所有前台进程  终止
- SIGQUIT 由 QUIT 字符(通常是Ctrl+\)产生，与 SIGINT 类似  终止
- SIGILL  进程企图执行非法指令时产生  终止
- SIGSEGV 非法访问内存时产生，如野指针、缓冲区溢出  终止
- SIGPIPE  进程往没有读端的管道中写入时产生，代表 “管道断裂”   终止
- SIGKILL  用来结束进程，且不能被捕捉和忽略   终止  
- SIGSTOP  用于暂停进程，且不能被捕捉和忽略   暂停进程 
- SIGTSTP  用于暂停进程，用户可键入 SUSP 字符（通常是 Ctrl+Z）发出此信号    暂停进程
- SIGCONT  让进程进入运行态  继续运行
- SIGALRM  通知进程定时器时间已到   终止
- SIGUSR1/2  保留给用户程序使用   终止
- SIGCHLD  当子进程退出时，给父进程发送该信号   忽略

## signal函数

- 头文件

  #include <signal.h>

- 原型

  typedef void (*sighandler_t)(int);
  sighandler_t signal(int signum, sighandler_t handler);

- 功能

  建立信号与处理方式的关联

- 参数

  signum 信号的编号

  handler 处理方式

  > SIG_IGN : 忽略
  > SIG_DFL ：默认
  > 传递函数地址：捕捉

- 返回值

  成功：返回handler首地址
  失败：返回SIG_ERR，重置错误码

## raise函数

- 头文件

  #include <signal.h>

- 原型

  int raise(int sig);

- 功能

  会向自身发送指定的信号

- 参数

  sig 要发送的信号的编号

- 返回值

  成功：返回0
  失败：返回非0

## kill函数

- 头文件

  #include <sys/types.h>
  #include <signal.h>

- 原型

  int kill(pid_t pid, int sig);

- 功能

  给任意进程发信号

- 参数

  pid 发送信号目标

  >\>0 ：向pid对应的单个进程发送信号(最常用场景)
  >0 ：向同进程组的所有进程发送信号
  >-1 : 向所有有权限操作的进程发送信号(init进程除外，避免影响系统核心进程)
  ><-1 : 给进程组ID等于-pid的进程发信号

  sig 信号的编号

- 返回值

  成功：0
  失败：-1，重置错误码

## atoi函数

- 头文件

  #include <stdlib.h>

- 原型

  int atoi(const char *nptr);

- 功能

  将字符串转换为整数

- 参数

  nptr 要转换的字符串的首地址

- 返回值

  成功：返回转换后的整数
  失败：返回0；如果转换后整数超出int范围，那么行为未定义；

## 示例代码

```c
// 模拟kill命令
#include <stdio.h>
#include <sys/types.h>
#include <signal.h>
#include <stdlib.h>


int main(int argc, const char *argv[])
{
    if (argc != 3 || argv[1][0] != '-')
    {
        printf("Please check arguments\n");
        return -1;
    }
    int signalno = 0;
    int pid = 0;
    signalno = atoi(argv[1] + 1);
    if (signalno == 0)
    {
        perror("Please check signal number");
        return -1;
    }
    pid = atoi(argv[2]);
    if (pid == 0)
    {
        perror("Please check PID");
        return -1;
    }
    int kill_return = 0;
    kill_return = kill(pid, signalno);
    if (kill_return == -1)
    {
        perror("Kill failed");
        return -1;
    }
    return 0;
}
```

# 消息队列

​	消息队列是内核维护的一种进程间通信(IPC)机制；如进程A可向消息队列写入带有【类型标识】和【消息正文】的消息；进程B则能根据预设的类型筛选条件，从队列中读取对应类型的消息(读取后消息从队列中移除)；
​	linux系统下，消息队列默认总大小通常是**6K**

![消息队列](/../assets/pictures/消息队列.png)

## System V IPC 对象和常用命令

- ipcs 显示指定System V IPC对象信息

  -q	显示消息队列相关信息

  -m	显示共享内存段相关信息

  -s	  显示信号量的相关信息

  -a	  显示所有类型的详细信息

- ipcrm 删除指定System V IPC对象信息

  -q	删除消息队列	ipcrm -q 5

  -m	删除共享内存段相关信息	ipcrm -m 10

  -s	  删除信号量的相关信息	ipcrm -s 15

## ftok函数

- 头文件

  #include <sys/types.h>
  #include <sys/ipc.h>

- 原型

  key_t ftok(const char *pathname, int proj_id);

- 功能

  用于生成一个key_t类型的键值，专门供System V系列的IPC函数使用
  作为不同进程识别同一IPC资源的“唯一标识”

- 参数

  pathname 必须指向一个已存在且可访问的文件
  proj_id 用于区分同一个文件关联的不同IPC资源的“项目标识符”(proj_id & 0xFF)

- 返回值

  成功：返回生成的key_t类型键值
  失败：返回-1，重置错误码(ENOENT:文件不存在；EASSES：权限不足)

## msgget函数

- 头文件

  #include <sys/types.h>
  #include <sys/ipc.h>
  #include <sys/msg.h>

- 原型

  int msgget(key_t key, int msgflg);

- 功能

  创建或获取一个消息队列
  若key对应的消息队列不存在，且msgflg包含IPC_CREAT标志，则创建新的消息队列；
  若key对应的消息队列存在，则获取该消息队列的标识符；

- 参数

  key 键值(通过ftok函数获取)，若是IPC_PRIVATE表示只能用于亲缘进程
  msgflg 消息队列的标志位(通常需结合权限位进行设置消息队列的访问权限)

  相关宏

  > IPC_CREAT    msgget(key, IPC_CREAT | 0666);
  >
  > 如果消息队列不存在,则创建新的消息队列；
  > 如果消息队列存在,则返回已存在消息队列
  > 的标识符,而不重新创建；
  >
  > IPC_EXCL    msgget(key,IPC_CREAT | IPC_EXCL | 0666);
  >
  > 必须与IPC_CREAT一起使用；
  > 若消息队列存在，msgget函数将失败，
  > 被设置为EEXIST;

- 返回值

  成功：返回消息队列的id
  失败：返回-1，重置错误码

## msgsnd函数

- 头文件

  #include <sys/types.h>
  #include <sys/ipc.h>
  #include <sys/msg.h>

- 原型

  int msgsnd(int msqid, const void *msgp, size_t msgsz, int msgflg);

- 功能

  向消息队列中写入一条消息

- 参数

  msqid 消息队列的id
  msgp 要写入数据的首地址

  > struct msgbuf {
  > long mtype; /* 消息类型，必须大于0 */
  > char mtext[1]; /* 消息正文，可以自定义 */
  > };

  msgsz 消息正文的大小
  msgflg 标志位(0：阻塞发送；IPC_NOWAIT：非阻塞发送)

- 返回值

  成功：返回0
  失败：返回-1，重置错误码

## msgrcv函数

- 头文件

  #include <sys/types.h>
  #include <sys/ipc.h>
  #include <sys/msg.h>

- 原型

  ssize_t msgrcv(int msqid, void *msgp, size_t msgsz, long msgtyp, int msgflg);

- 功能

  从消息队列中读取一条消息

- 参数

  msqid 消息队列的id
  msgp 用于保存接收的数据的缓冲区的首地址

  > struct msgbuf {
  > long mtype; /* 消息类型，必须大于0 */
  > char mtext[1]; /* 消息正文，可以自定义 */
  > };

  msgsz 消息正文的大小

  msgtyp 要接收的消息的类型

  > 0 ：接收消息队列中的第一条消息
  >
  > \>0：接收指定类型的第一条消息(最常用)
  > <0：一般不使用，表示接收消息队列中第一条类型最小的小于msgtyp的绝对值的消息msgflg 标志位(0：阻塞发送；IPC_NOWAIT：非阻塞发送)

  msgflg 标志位(0：阻塞接收；IPC_NOWAIT：非阻塞接收)

- 返回值

  成功：返回实际读到的正文的字节数
  失败：返回-1，重置错误码

## msgctl函数

- 头文件

  #include <sys/types.h>
  #include <sys/ipc.h>
  #include <sys/msg.h>

- 原型

  int msgctl(int msqid, int cmd, struct msqid_ds *buf);

- 功能

  控制消息队列

- 参数

  msqid 消息队列的id
  cmd 指令

  > 指令相关宏
  >
  > IPC_RMID 删除消息队列msgctl(msgid, IPC_RMID, NULL);
  > IPC_SET 设置消息队列属性 msgctl(msgid, IPC_SET, &buf);
  > IPC_STAT 获取消息队列属性msgctl(msgid, IPC_STAT, &buf);

  buf 属性结构体的地址

- 返回值

  成功：0
  失败：返回-1，重置错误码

## msqid_ds结构体

```c
struct msqid_ds
{
    struct ipc_perm msg_perm;   /* IPC权限结构体 */
    time_t msg_stime;           /* 最后一次执行msgsnd的时间 */
    time_t msg_rtime;           /* 最后一次执行msgrcv的时间 */
    time_t msg_ctime;           /* 最后一次被修改的时间 */
    unsigned long __msg_cbytes; /* 当前消息队列中的字节数 */
```

# 共享内存

在内核中创建共享内存，让进程都可以访问到，通过这段内存进行数据的传递

共享内存是所有进程间通信方式中效率最高的(不需要来回进行数据拷贝)；

![共享内存](/../assets/pictures/共享内存.png)

## shmget函数

- 头文件

  #include <sys/ipc.h>
  #include <sys/shm.h>

- 原型

  int shmget(key_t key, size_t size, int shmflg)

- 功能

  创建新的共享内存段或获取已存在的共享内存段的标识符

- 参数

  key 键值，通过ftok获取(IPC_PRIVATE用于亲缘进程间的通信)
  size 共享内存的大小，单位字节, PAGE_SIZE(4k)的整数倍,如果小于PAGE_SIZE, 则实际大小为PAGE_SIZE但只能访问size个
  shmflg 共享的标志位, 和消息队列的标志宏一样

- 返回值

  成功：返回共享内存编号
  失败：返回-1，重置错误码

## shmat函数

- 头文件

  #include <sys/types.h>
  #include <sys/shm.h>

- 原型

  void *shmat(int shmid, const void *shmaddr, int shmflg);

- 功能

  映射共享内存到当前的进程空间

- 参数

  shmid 共享内存的编号
  shmaddr 指向共享内存的地址(若是NULL，让系统自动分配)
  shmflg

  > 用于指定共享内存的访问权限
  > 0 ：默认权限-读写权限
  > SHM_RDONLY：只读权限

- 返回值

  成功：指向共享内存的地址
  失败：返回(void *)-1，重置错误码

## shmdt函数

- 头文件

  #include <sys/types.h>
  #include <sys/shm.h>

- 原型

  int shmdt(const void *shmaddr);

- 功能

  取消地址映射

- 参数

​	shmaddr 指向共享内存的地址

- 返回值

​	成功：0
​	失败：返回-1，重置错误码

## shmctl函数

- 头文件

  #include <sys/ipc.h>
  #include <sys/shm.h>

- 原型

  int shmctl(int shmid, int cmd, struct shmid_ds *buf);

- 功能

  共享内存控制的函数

- 参数

  shmid 共享内存编号
  cmd

  > IPC_STAT：获取共享内存
  > IPC_SET ：设置共享内存
  > IPC_RMID: 删除共享内存

​	buf 共享内存属性结构体指针

- 返回值

​	成功：0
​	失败：返回-1，重置错误码

# 信号量集

信号量集又叫做是信号灯集，它是实现进程间同步的机制，在一个信号量集中可以有多个信号量，这些信号量之间工作互不干扰；一般使用的都是二值信号量

![信号量集](/../assets/pictures/信号量集.png)

## semget函数

- 头文件

  #include <sys/types.h>
  #include <sys/ipc.h>
  #include <sys/sem.h>

- 原型

  int semget(key_t key, int nsems, int semflg);

- 功能

  创建一个信号量集
  若信号量集不存在且semflg包含IPC_CREAT，则创建信号量集
  若信号量集已存在，则获取该信号量集的标识符

- 参数

  key 键值(通过ftok获取)
  nsems 信号量集合中信号量的个数
  semflg 创建的标志位, 和消息队列, 共享内存中的宏一样

- 返回值

  成功：semid
  失败：返回-1，重置错误码

## semctl函数

- 头文件

  #include <sys/types.h>
  #include <sys/ipc.h>
  #include <sys/sem.h>

- 原型

  int semctl(int semid, int semnum, int cmd, ...);

- 功能

  信号量的控制函数

- 参数

  semid 信号量集的标识符
  semnum 信号量的编号(从0开始)
  cmd 命令码

  > SETVAL 设置单个信号量的值，需传递第四个参数
  > GETVAL 获取单个信号量的值，无需传递第四个参数
  > IPC_STAT 获取信号量集的整体属性，此时semnum被忽略，需传递buf指针存储属性
  > IPC_SET 设置信号量集的整体属性，此时semnum被忽略，需传递buf指针指定新属性
  > IPC_RMID 删除整个信号量集，此时semnum被忽略，无需传递第四个参数

  ... 可变参数


- 返回值

  成功：
  GETVAL：成功返回信号量的值
  其余的命令码成功返回0
  失败：返回-1，重置错误码

## semop函数

- 头文件

  #include <sys/types.h>
  #include <sys/ipc.h>
  #include <sys/sem.h>

- 原型

  int semop(int semid, struct sembuf *sops, size_t nsops);

- 功能

  用于对信号量集中的一个或多个信号量执行原子操作(如 P/V 操作)

- 参数

  semid 信号量集的标识符
  sops 操作方式

  > struct sembuf{
  > unsigned short sem_num; //信号量的编号
  > short sem_op; //操作方式
  > short sem_flg; //操作的标志位
  > }

​	nsops 本次操作信号量的个数

- 返回值

  成功：0
  失败：返回-1，重置错误码

## P/V操作

- P 操作申请资源S = S - 1
  S ≥ 0：申请成功，进程继续(占用 1 份资源)；
  S < 0：申请失败，进程阻塞等待；
- V 操作释放资源S = S + 1
  S > 0：释放成功，无进程等待(资源剩余增加 1 份)
  S ≤ 0：释放后唤醒 1 个阻塞进程(给等待队列分配 1 份资源)

## 操作标志sem_flg

- 0 	

  默认值(阻塞模式) 	进程可容忍等待资源(如正常业务逻辑中申请共享内存访问权限)

- IPC_NOWAIT       

  非阻塞模式        进程无法等待(如实时任务,若资源不可用需立即返回并处理其他逻辑)，此时操作无法完成会返回-1，errno=EAGAIN

- SEM_UNDO        

  操作回滚(进程异常退出时恢复信号量状态）      防止进程因崩溃导致信号量 “卡死”如进程执行 P 操作后未执行 V 操作就退出，SEM_UNDO会自动将S加 1，释放资源;

```c
        }

        return -1;

    }

    // 创建key
    key_t key = ftok(argv[1], 1);
    if (key == -1)
    {
        perror("Create key failed");
        return -1;
    }
    printf("Create key succeed, key = %x\n", key);

    // 创建共享内存
    int shm_id = shmget(key, SHM_SIZE, IPC_CREAT | 0666);
    if (shm_id == -1)
    {
        perror("Create shared memory faild");
    }
    printf("Create shared memory succeed, shm_id = %d\n", shm_id);

    // 映射共享内存到进程
    char *shm_address = NULL;
    shm_address = shmat(shm_id, NULL, SHM_RDONLY);
    if (shm_address == (void *)-1)
    {
        perror("Attach shared memory faild");
        return -1;
    }
    printf("Attach shared memory succeed, shm_address = %x\n", shm_address);

    // 创建信号量集
    int sem_id = semget(key, 2, IPC_CREAT | 0666);
    if (sem_id == -1)
    {
        perror("Create sem faild");
        return -1;
    }
    printf("Create sem succeed, sem_id = %d\n", shm_id);

    // 通过信号量阻塞进行读写
    while (1)
    {
        // 申请读信号量
        int semp_return = sem_p(sem_id, 1);
        if (semp_return == -1)
        {
            shmdt(shm_address);
            return -1;
        }
        if (!strcmp(shm_address, "quit"))
        {
            break;
        }
        printf("Message readed:%s\n", shm_address);
        // 释放写信号量
        semp_return = sem_v(sem_id, 0);
        if (semp_return == -1)
        {
            shmdt(shm_address);
            return -1;
        }
    }


    // 取消映射共享内存到进程
    int shmdt_return = shmdt(shm_address);
    if (shmdt_return == -1)
    {
        perror("Dettach shared memory faild");
        return -1;
    }
    printf("Dettach shared memory succeed\n");

    // 删除共享内存
    int shmctl_return = shmctl(shm_id, IPC_RMID, NULL);
    if (shmctl_return == -1)
    {
        perror("Release shared memory faild");
        return -1;
    }
    printf("Release shared memory succeed\n");
    
    // 删除信号量集
    int semctl_return = semctl(sem_id, 0, IPC_RMID);
    if (semctl_return == -1)
    {
        perror("Release sem faild");
        return -1;
    }
    printf("Release sem succeed\n");
    return 0;
}
```

# 进程间通信方式使用场景

| 通信方式 | 典型使用场景                                                 | 适配情况                             |
| -------- | ------------------------------------------------------------ | ------------------------------------ |
| 无名管道 | 1. 命令行工具协作(如 Linux 中 ls \|grep “hello”，前进程输出作为后进程输入)；<br />2. 父进程与子进程的简单数据传递(如子进程向父进程返回日志)； | 只能用于亲缘进程通信                 |
| 有名管道 | 1. 同一主机内独立服务的通信(如桌面应用向后台服务发送控制指令，如 “暂停下载”)；<br/>2. 嵌入式系统中不同进程的低延迟数据交互(如传感器进程向处理进程传数据)； | 一般用于非亲缘进程通信, 保证传输效率 |
| 信号     | 1. 进程间简单控制指令(如父进程发送 SIGTERM 终止子进程)；<br/>2. 系统级事件响应(如 Ctrl+C 触发 SIGINT 中断前台进程，或硬件错误触发 SIGSEGV 终止崩溃进程)； | 用于发送通知, 处理异常时间           |
| 消息队列 | 1. 高并发场景下的异步通信(如订单系统进程向库存系统进程发送 “扣减库存” 请求，无需等待响应)；<br/>2. 进程间需按优先级处理数据(如系统日志进程优先处理 “错误日志”，再处理 “-info 日志”)； | info日志, 用来记录运行时的关键事件   |
| 共享内存 | 1. 超高频、大数据量交互(如视频编辑软件的 “预览进程” 与 “渲染进程” 共享视频帧数据，避免拷贝开销)；<br/>2. 实时系统(如工业控制中，传感器数据进程与控制逻辑进程共享实时采集数据)； | 直接操作内存, 效率高                 |
| 信号量集 | 1. 共享资源互斥访问(如多个进程竞争同一文件，用信号量保证 “同一时间仅一个进程写入”)；<br/>2. 进程协作(如 “生产者进程” 生产数据后发信号量，“消费者进程” 等待信号量后读取数据)； | 二值, 资源控制                       |
