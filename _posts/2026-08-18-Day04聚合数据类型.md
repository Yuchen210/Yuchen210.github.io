# 结构体

- 定义

  ```c
  struct结构体名 {
  	数据类型 成员名1;
  	数据类型 成员名2;
  	// 其他成员
  };
  
  //定义结构体变量
  struct Person person1;
  person1.age = 25;
  strcpy(person1.name, "Alice");
  persion1.height = 1.65;
  
  //基本定义
  struct Student{
      int age;
      char name[50];
  }
  struct Student stu1;
  
  //定义同时声明变量, 节省代码空间
  struct Student{
      int age;
      char name[50];
  }stu1, stu2;
  
  //结构体类型别名
  typedef struct{
      int age;
      char name[50];
  }Student;
  Student stu1;
  ```

- 定义一个学生结构体, 包含学生的学号, 姓名和三门课程的成绩, 输入5个学生的信息并输出每个学生的总成绩和平均成绩

  ```c
  #include <stdio.h>
  #include <stdlib.h>
  #include <stdbool.h>
  #include <string.h>
  
  typedef struct{
      float math;
      float chinese;
      float english;
  }Score;
  
  typedef struct{
      char name[20];
      Score score;
  }Student;
  
  int main(int argc, char *argv[])
  {
      Student stu[5];
      for(int i = 0; i < 5; i++){
          printf("请输入第%d个学生的姓名：", i+1);
          scanf("%s", stu[i].name);
          printf("请输入第%d个学生的数学成绩：", i+1);
          scanf("%f", &stu[i].score.math);
          printf("请输入第%d个学生的语文成绩：", i+1);
          scanf("%f", &stu[i].score.chinese);
          printf("请输入第%d个学生的英语成绩：", i+1);
          scanf("%f", &stu[i].score.english);
      }
      for(int i = 0; i < 5; i++){
          printf("姓名：%s，数学成绩：%.2f，语文成绩：%.2f，英语成绩：%.2f, 总成绩：%.2f, 平均成绩：%.2f\n", stu[i].name, stu[i].score.math, stu[i].score.chinese, stu[i].score.english, stu[i].score.math + stu[i].score.chinese + stu[i].score.english, (stu[i].score.math + stu[i].score.chinese + stu[i].score.english) / 3);
      }
      return 0;
  }
  ```

- 定义一个Book结构体,包含书名, 作者, 价格, 出版年份等信息, 输入3本书并按价格升序输出

  ```c
  #include <stdio.h>
  #include <stdlib.h>
  #include <stdbool.h>
  #include <string.h>
  
  typedef struct{
      char bookName[20];
      char author[20];
      float price;
      int year;
  }bookBook;
  
  void copybook(bookBook *book1, bookBook *book2){
      strcpy(book1->author, book2->author);
      strcpy(book1->bookName, book2->bookName);
      book1->price = book2->price;
      book1->year = book2->year;
  }
  
  void switchbook(bookBook *book1, bookBook *book2){
      bookBook temp;
      copybook(&temp, book1);
      copybook(book1, book2);
      copybook(book2, &temp);
  }
  
  int main(int argc, char *argv[])
  {
      bookBook book[3];
      for(int i = 0; i < 3; i++){
          scanf("%s %s %f %d", book[i].bookName, book[i].author, &book[i].price, &book[i].year);
      }
      if(book[0].price > book[1].price){
          switchbook(&book[0], &book[1]);
      }
      if(book[1].price > book[2].price){
          switchbook(&book[1], &book[2]);
      }
      if(book[0].price > book[1].price){
          switchbook(&book[0], &book[1]);
      }
      for(int i = 0; i < 3; i++){
          printf("%s %s %f %d\n", book[i].bookName, book[i].author, book[i].price, book[i].year);
      }
      return 0;
  }
  ```
